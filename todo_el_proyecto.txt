package com.capuccinolovesballerina.game.entidades;

import com.capuccinolovesballerina.game.Utilidades.Constantes;
import com.capuccinolovesballerina.game.objetos.BloqueEspuma;

public class BallerinaCapuccina extends Player {

    private BloqueEspuma espuma = null;

    private float cooldownEspuma = 0f;

    public BallerinaCapuccina(float x, float y) {
        super(x, y, Constantes.ANCHO_PERSONAJE, Constantes.ALTO_PERSONAJE);
    }

    public boolean crearEspuma() {
        if (cooldownEspuma > 0f) {
            return false;
        }

        espuma = null;

        float ancho = Constantes.ANCHO_ESPUMA;
        float alto = Constantes.ALTO_ESPUMA;
        float gap = Constantes.DISTANCIA_ESPUMA;

        float xEspuma;
        if (isMirandoDerecha()) {
            xEspuma = getX() + getAncho() + gap;
        } else {
            xEspuma = getX() - gap - ancho;
        }

        float yEspuma;
        if(enSuelo) {
            yEspuma = getY();
        }else{
            yEspuma = getY() - alto;
        }

        espuma = new BloqueEspuma(xEspuma, yEspuma, ancho, alto, Constantes.VIDA_ESPUMA);
        cooldownEspuma = Constantes.COOLDOWN_ESPUMA;
        return true;
    }

    public void actualizarHabilidad(float delta) {
        if (cooldownEspuma > 0f) {
            cooldownEspuma -= delta;
            if (cooldownEspuma < 0f) cooldownEspuma = 0f;
        }

        if (espuma != null) {
            espuma.actualizar(delta);
            if (!espuma.estaViva()) {
                espuma = null;
            }
        }
    }

    public BloqueEspuma getEspuma() {
        return espuma;
    }

    public void reiniciarHabilidad() {
        espuma = null;
        cooldownEspuma = 0f;
    }

    public void setEspumaRemota(float x, float y) {
        espuma = new BloqueEspuma(
            x,
            y,
            Constantes.ANCHO_ESPUMA,
            Constantes.ALTO_ESPUMA,
            Constantes.VIDA_ESPUMA
        );
        cooldownEspuma = 0f;
    }
}

package com.capuccinolovesballerina.game.entidades;

import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;
import com.capuccinolovesballerina.game.Utilidades.Constantes;

public class Player {


    protected float x;
    protected float y;
    protected float ancho;
    protected float alto;

    protected float velocidadX;
    protected float velocidadY;

    protected boolean enSuelo;
    protected boolean mirandoDerecha;

    private final Rectangle caja = new Rectangle();

    public Player() {
        this(0, 0, Constantes.ANCHO_PERSONAJE, Constantes.ALTO_PERSONAJE);
    }

    public Player(float x, float y, float ancho, float alto) {
        this.x = x;
        this.y = y;
        this.ancho = ancho;
        this.alto = alto;
        this.mirandoDerecha = true;
    }

    public void actualizar(float delta, boolean izquierda, boolean derecha, boolean saltar, Array<Rectangle> solidos) {

        float direccion = 0;

        if (izquierda) {
            direccion = -1;
        }

        if (derecha) {
            direccion = 1;
        }

        velocidadX = direccion * Constantes.VELOCIDAD_MOVIMIENTO;

        if (direccion > 0) {
            mirandoDerecha = true;
        }

        if (direccion < 0) {
            mirandoDerecha = false;
        }

        if (saltar && enSuelo) {
            velocidadY = Constantes.VELOCIDAD_SALTO;
            enSuelo = false;
        }

        velocidadY -= Constantes.GRAVEDAD * delta;

        if (velocidadY < -Constantes.VELOCIDAD_MAX_CAIDA) {
            velocidadY = -Constantes.VELOCIDAD_MAX_CAIDA;
        }

        moverEnX(delta, solidos);
        moverEnY(delta, solidos);
    }

    private void moverEnX(float delta, Array<Rectangle> solidos) {

        x += velocidadX * delta;
        caja.set(x, y, ancho, alto);

        for (Rectangle solido : solidos) {
            if (caja.overlaps(solido)) {

                if (velocidadX > 0) {
                    x = solido.x - ancho;
                } else if (velocidadX < 0) {
                    x = solido.x + solido.width;
                }

                caja.set(x, y, ancho, alto);
                velocidadX = 0;
            }
        }
    }

    private void moverEnY(float delta, Array<Rectangle> solidos) {

        y += velocidadY * delta;
        caja.set(x, y, ancho, alto);

        enSuelo = false;

        for (Rectangle solido : solidos) {
            if (caja.overlaps(solido)) {

                if (velocidadY < 0) {
                    y = solido.y + solido.height;
                    enSuelo = true;
                } else if (velocidadY > 0) {
                    y = solido.y - alto;
                }

                caja.set(x, y, ancho, alto);
                velocidadY = 0;
            }
        }
    }

    public Rectangle getCaja() {
        caja.set(x, y, ancho, alto);
        return caja;
    }

    public float getX() {
        return x;
    }

    public float getY() {
        return y;
    }

    public float getAncho() {
        return ancho;
    }

    public float getAlto() {
        return alto;
    }

    public boolean isEnSuelo() {
        return enSuelo;
    }

    public boolean isMirandoDerecha() {
        return mirandoDerecha;
    }

    public void setX(float x) {
        this.x = x;
    }

    public void setY(float y) {
        this.y = y;
    }

    public void reaparecer(float x, float y) {
        this.x = x;
        this.y = y;
        this.velocidadX = 0;
        this.velocidadY = 0;
        this.enSuelo = false;
    }

    public void setPosicionRemota(float x, float y, boolean mirandoDerecha, boolean enSuelo) {
        this.x = x;
        this.y = y;
        this.velocidadX = 0f;
        this.velocidadY = 0f;
        this.mirandoDerecha = mirandoDerecha;
        this.enSuelo = enSuelo;
    }
}

package com.capuccinolovesballerina.game.graficos;

import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.Animation;
import com.badlogic.gdx.graphics.g2d.TextureRegion;

public class AnimacionesCappuccino {



    private static final int FRAME = 128;
    private static final int FRAMES_IDLE = 4;
    private static final int FRAMES_WALK = 9;
    private static final int FRAMES_JUMP = 3;
    private static final int FRAMES_ATTACK = 7;

    private final Texture textura;
    private final Animation<TextureRegion> idle;
    private final Animation<TextureRegion> walk;
    private final Animation<TextureRegion> jump;
    private final Animation<TextureRegion> attack;

    private Estado estado = Estado.IDLE;
    private float tiempo = 0f;

    public AnimacionesCappuccino() {
        textura = new Texture("sprites/cappuccino.png");

        TextureRegion[][] grilla = TextureRegion.split(textura, FRAME, FRAME);

        idle   = new Animation<>(0.2f,  primeros(grilla[0], FRAMES_IDLE));
        idle.setPlayMode(Animation.PlayMode.LOOP);

        walk   = new Animation<>(0.1f,  primeros(grilla[1], FRAMES_WALK));
        walk.setPlayMode(Animation.PlayMode.LOOP);

        jump   = new Animation<>(0.15f, primeros(grilla[2], FRAMES_JUMP));
        jump.setPlayMode(Animation.PlayMode.NORMAL);

        attack = new Animation<>(0.07f, primeros(grilla[3], FRAMES_ATTACK));
        attack.setPlayMode(Animation.PlayMode.NORMAL);
    }


    private TextureRegion[] primeros(TextureRegion[] fila, int n) {
        TextureRegion[] frames = new TextureRegion[n];
        for (int i = 0; i < n; i++) frames[i] = fila[i];
        return frames;
    }


    public void setEstado(Estado nuevo) {
        if (estado == Estado.ATTACK && !attack.isAnimationFinished(tiempo)) return;
        if (nuevo != estado) {
            estado = nuevo;
            tiempo = 0f;
        }
    }

    public void actualizar(float delta) {
        tiempo += delta;
    }

    public void reiniciar() {
        estado = Estado.IDLE;
        tiempo = 0f;
    }


    public TextureRegion getFrameActual() {
        switch (estado) {
            case WALK:   return walk.getKeyFrame(tiempo);
            case JUMP:   return jump.getKeyFrame(tiempo);
            case ATTACK: return attack.getKeyFrame(tiempo);
            default:     return idle.getKeyFrame(tiempo);
        }
    }

    public void dispose() {
        textura.dispose();
    }

package com.capuccinolovesballerina.game.mapa;

import com.badlogic.gdx.maps.MapObject;
import com.badlogic.gdx.maps.objects.RectangleMapObject;
import com.badlogic.gdx.maps.tiled.TiledMap;
import com.badlogic.gdx.maps.tiled.TiledMapTile;
import com.badlogic.gdx.maps.tiled.TiledMapTileLayer;
import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;


public class GestorColisiones {

    private final Array<Rectangle> solidos = new Array<>();

    /**
     * @param tamanoTile tamano del tile en pixeles de Tiled (por ejemplo 32)
     * @param escala     factor para pasar de pixeles de Tiled a unidades del mundo (Nivel.ESCALA)
     */
    public GestorColisiones(TiledMap mapa, int tamanoTile, float escala) {
        for (TiledMapTileLayer capa : mapa.getLayers().getByType(TiledMapTileLayer.class)) {
            if (capa.getName().toLowerCase().startsWith("fondo")) continue;
            for (int x = 0; x < capa.getWidth(); x++) {
                for (int y = 0; y < capa.getHeight(); y++) {
                    TiledMapTileLayer.Cell celda = capa.getCell(x, y);
                    if (celda == null) continue; // celda vacia

                    TiledMapTile tile = celda.getTile();

                    for (MapObject obj : tile.getObjects()) {
                        if (obj instanceof RectangleMapObject) {
                            Rectangle r = ((RectangleMapObject) obj).getRectangle();

                            // Se calcula en pixeles de Tiled y despues se escala todo junto.
                            float mundoX = x * tamanoTile + r.x;
                            float mundoY = y * tamanoTile + (tamanoTile - r.y - r.height);
                            solidos.add(new Rectangle(
                                mundoX * escala,
                                mundoY * escala,
                                r.width * escala,
                                r.height * escala
                            ));
                        }
                    }
                }
            }
        }
    }

    public Array<Rectangle> getSolidos() {
        return solidos;
    }

    public boolean colisiona(Rectangle entidad) {
        for (Rectangle r : solidos) {
            if (r.overlaps(entidad)) return true;
        }
        return false;
    }
}

package com.capuccinolovesballerina.game.mapa;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.maps.MapLayer;
import com.badlogic.gdx.maps.MapObject;
import com.badlogic.gdx.maps.MapProperties;
import com.badlogic.gdx.maps.objects.RectangleMapObject;
import com.badlogic.gdx.maps.tiled.TiledMap;
import com.badlogic.gdx.maps.tiled.TmxMapLoader;
import com.badlogic.gdx.maps.tiled.objects.TiledMapTileMapObject;
import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;

public class Nivel {

    /**
     * Escala entre "pixeles de Tiled" y "unidades del mundo".
     * 1f = el mundo mide lo mismo que el mapa en Tiled (30x20 tiles de 32 = 960x640).
     * Es como esta disenado NIVEL 0 (puerta 96x128, peligro 64x32, etc.).
     * Si algun dia queres un mundo mas grande sin retocar el mapa, cambia SOLO este numero.
     */
    public static final float ESCALA = 1f;

    /** true = los tiles se dibujan sin suavizar (pixel art). */
    private static final boolean PIXEL_ART = false;

    /** Dejalo en true hasta que todo funcione; despues ponelo en false. */
    private static final boolean DEBUG_OBJETOS = true;

    private final TiledMap mapa;
    private final GestorColisiones colisiones;
    private final Array<MapObject> objetos = new Array<>();
    private final float anchoPx;
    private final float altoPx;

    public Nivel(String ruta) {
        TmxMapLoader.Parameters parametros = new TmxMapLoader.Parameters();
        if (PIXEL_ART) {
            parametros.generateMipMaps = false;
            parametros.textureMinFilter = Texture.TextureFilter.Nearest;
            parametros.textureMagFilter = Texture.TextureFilter.Nearest;
        }
        mapa = new TmxMapLoader().load(ruta, parametros);

        MapProperties props = mapa.getProperties();
        int tileAncho = props.get("tilewidth", Integer.class);
        int tileAlto = props.get("tileheight", Integer.class);
        int tilesX = props.get("width", Integer.class);
        int tilesY = props.get("height", Integer.class);

        anchoPx = tilesX * tileAncho * ESCALA;
        altoPx = tilesY * tileAlto * ESCALA;

        // Se leen los objetos de TODAS las capas de objetos, se llamen como se llamen
        // (Tiled las crea como "Capa de Objetos 1" por defecto).
        for (MapLayer capa : mapa.getLayers()) {
            for (MapObject obj : capa.getObjects()) {
                objetos.add(obj);
            }
        }

        colisiones = new GestorColisiones(mapa, tileAncho, ESCALA);

        if (DEBUG_OBJETOS) {
            logObjetos();
        }
    }

    public TiledMap getMapa() {
        return mapa;
    }

    public GestorColisiones getColisiones() {
        return colisiones;
    }

    /** Ancho total del mapa en unidades del mundo. */
    public float getAnchoPx() {
        return anchoPx;
    }

    /** Alto total del mapa en unidades del mundo. */
    public float getAltoPx() {
        return altoPx;
    }

    /** Primer objeto con ese nombre/type/class, en unidades del mundo (o null). */
    public Rectangle buscarObjeto(String tipo) {
        for (MapObject obj : objetos) {
            if (!coincide(obj, tipo)) continue;
            Rectangle r = rectanguloDe(obj);
            if (r != null) return r;
        }
        if (DEBUG_OBJETOS) {
            Gdx.app.log("NIVEL", "NO se encontro el objeto '" + tipo + "' (revisa name/type en Tiled)");
        }
        return null;
    }

    /** Todos los objetos con ese nombre/type/class, en unidades del mundo. */
    public Array<Rectangle> buscarObjetos(String tipo) {
        Array<Rectangle> resultado = new Array<>();
        for (MapObject obj : objetos) {
            if (!coincide(obj, tipo)) continue;
            Rectangle r = rectanguloDe(obj);
            if (r != null) resultado.add(r);
        }
        return resultado;
    }

    public void dispose() {
        mapa.dispose();
    }

    // ---------------------------------------------------------------- internos

    /** Compara con name, type o class (Tiled 1.9+), ignorando mayusculas y espacios. */
    private static boolean coincide(MapObject obj, String tipo) {
        MapProperties p = obj.getProperties();
        return igual(tipo, obj.getName()) || igual(tipo, p.get("type")) || igual(tipo, p.get("class"));
    }

    private static boolean igual(String tipo, Object valor) {
        return valor != null && tipo.equalsIgnoreCase(valor.toString().trim());
    }

    /** Convierte el objeto de Tiled a un rectangulo en unidades del mundo. */
    private static Rectangle rectanguloDe(MapObject obj) {
        if (obj instanceof RectangleMapObject) {
            Rectangle r = ((RectangleMapObject) obj).getRectangle();
            return new Rectangle(r.x * ESCALA, r.y * ESCALA, r.width * ESCALA, r.height * ESCALA);
        }
        // Objetos puestos con "Insertar tile"
        if (obj instanceof TiledMapTileMapObject) {
            MapProperties p = obj.getProperties();
            return new Rectangle(
                numero(p, "x") * ESCALA,
                numero(p, "y") * ESCALA,
                numero(p, "width") * ESCALA,
                numero(p, "height") * ESCALA
            );
        }
        return null;
    }

    private static float numero(MapProperties p, String clave) {
        Object v = p.get(clave);
        return v instanceof Number ? ((Number) v).floatValue() : 0f;
    }

    private void logObjetos() {
        Gdx.app.log("NIVEL", "Mundo: " + anchoPx + " x " + altoPx + " (escala " + ESCALA + "), "
            + objetos.size + " objetos");
        for (MapObject obj : objetos) {
            MapProperties p = obj.getProperties();
            Gdx.app.log("NIVEL", obj.getClass().getSimpleName()
                + " name=" + obj.getName()
                + " type=" + p.get("type")
                + " x=" + p.get("x") + " y=" + p.get("y")
                + " w=" + p.get("width") + " h=" + p.get("height"));
        }
    }
}

        shapeRenderer.begin(ShapeRenderer.ShapeType.Filled);
        shapeRenderer.setColor(0, 0, 0, 0.6f);
        shapeRenderer.rect(0, 0, viewport.getWorldWidth(), viewport.getWorldHeight());
        shapeRenderer.end();

        stage.draw();
    }

    public void resize(int width, int height) {
        viewport.update(width, height, true);
    }

    public void dispose() {
        stage.dispose();
        shapeRenderer.dispose();
    }
}

package com.capuccinolovesballerina.game.pantallas;

public class PantallaJuego implements Screen {

    private final Game juego;

    private OrthographicCamera camera;
    private Viewport viewport;

    private Nivel nivel;
    private ElementosNivel elementos;

    private MenuPausa menuPausa;
    private MenuGameOver menuGameOver;
    private MenuVictoria menuVictoria;
    private HUD hud;

    private ControladorEntradaJugador controladorEntrada;
    private CapuccinoAssassino cappuccino;
    private BallerinaCapuccina ballerina;
    private AnimacionesCappuccino animaciones;

    private ControladorPartida controladorPartida;
    private RenderizadorJuego renderizadorJuego;

    public PantallaJuego(Game juego) {
        this.juego = juego;
    }

    @Override
    public void show() {

        camera = new OrthographicCamera();

        nivel = new Nivel("mapas/NIVEL 0.tmx");
        viewport = FabricaViewport.crear(camera, nivel.getAnchoPx(), nivel.getAltoPx());
        elementos = new ElementosNivel(nivel);

        menuPausa = new MenuPausa(juego);
        menuGameOver = new MenuGameOver(juego, () -> controladorPartida.reiniciar());
        menuVictoria = new MenuVictoria(juego);
        hud = new HUD();

        controladorEntrada = new ControladorEntradaJugador();
        animaciones = new AnimacionesCappuccino();

        cappuccino = new CapuccinoAssassino(elementos.getSpawnX(), elementos.getSpawnY());
        ballerina = new BallerinaCapuccina(elementos.getSpawnBallerinaX(), elementos.getSpawnBallerinaY());

        Gdx.input.setInputProcessor(controladorEntrada);

        camera.position.set(nivel.getAnchoPx() / 2f, nivel.getAltoPx() / 2f, 0);
        camera.update();

        controladorPartida = new ControladorPartida(
            nivel, elementos, menuPausa, menuGameOver, menuVictoria,
            hud, controladorEntrada, cappuccino, ballerina, animaciones
        );

        renderizadorJuego = new RenderizadorJuego(
            camera, nivel, cappuccino, ballerina, animaciones,
            elementos, hud, menuPausa, menuGameOver, menuVictoria
        );

        Audio.reproducirMusica("sonidos/musica_juego.mp3");
    }

    @Override
    public void render(float delta) {
        float dt = Math.min(delta, 1f / 30f);
        controladorPartida.actualizar(dt);
        renderizadorJuego.dibujar();
    }

    @Override
    public void resize(int width, int height) {
        viewport.update(width, height, true);
        menuPausa.resize(width, height);
        menuGameOver.resize(width, height);
        menuVictoria.resize(width, height);
        hud.resize(width, height);
    }

    @Override
    public void dispose() {
        renderizadorJuego.dispose();
        nivel.dispose();
        menuPausa.dispose();
        menuGameOver.dispose();
        menuVictoria.dispose();
        hud.dispose();
        animaciones.dispose();
    }

    @Override
    public void pause() {}

    @Override
    public void resume() {}

    @Override
    public void hide() {
        Gdx.input.setInputProcessor(null);
    }
}

    private void prepararResolucionVentana() {

        if (Gdx.graphics.isFullscreen()) {
            return;
        }

        String resolucionActual = Gdx.graphics.getWidth() + " x " + Gdx.graphics.getHeight();

        if (esResolucionPermitida(resolucionActual)) {
            resolucionVentana = resolucionActual;
        } else {
            resolucionVentana = "1280 x 720";
            aplicarResolucionVentana(resolucionVentana);
        }

    private boolean esResolucionPermitida(String resolucion) {
        String normalizada = resolucion.replace(" ", "");
        return normalizada.equals("1280x720") || normalizada.equals("1024x768") || normalizada.equals("800x600");
    }

    private void aplicarResolucionVentana(String resolucion) {

    private boolean descubrirServidor() {
        DatagramSocket socket = null;
        try {
            socket = new DatagramSocket();
            socket.setBroadcast(true);
            socket.setSoTimeout(TIMEOUT_DESCUBRIMIENTO);

            byte[] pedido = "DISCOVER".getBytes();
            InetAddress broadcast = InetAddress.getByName("255.255.255.255");
            DatagramPacket dpOut = new DatagramPacket(pedido, pedido.length, broadcast, PUERTO_DESCUBRIMIENTO);
            socket.send(dpOut);

            byte[] buffer = new byte[64];
            DatagramPacket dpIn = new DatagramPacket(buffer, buffer.length);
            socket.receive(dpIn);

            String msg = new String(dpIn.getData(), 0, dpIn.getLength());
            String[] partes = msg.split("\\|", -1);
            if (partes.length < 2 || !"SERVER".equals(partes[0])) {
                return false;
            }
            ipServer = dpIn.getAddress().getHostAddress();
            puertoServer = Integer.parseInt(partes[1]);
            return true;
        } catch (Exception e) {
            return false;
        } finally {
            if (socket != null && !socket.isClosed()) {
                socket.close();
            }
        }
    }

    private boolean esperarRol() {
        long inicio = System.currentTimeMillis();
        while (miRol == -1) {
            for (String m : hc.drenar()) {
                String[] p = m.split("\\|", -1);
                if (p.length >= 2 && "ROL".equals(p[0])) {
                    miRol = Integer.parseInt(p[1]);
                } else {
                    pendientes.add(m);
                }
            }
            if (miRol != -1) {
                return true;
            }
            if (System.currentTimeMillis() - inicio > TIMEOUT_ROL) {
                return false;
            }
            try {
                Thread.sleep(50);
            } catch (InterruptedException e) {
                return false;
            }
        }
        return true;
    }

import java.io.IOException;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;
import java.net.SocketException;
import java.util.ArrayList;
import java.util.List;

public class HiloCliente extends Thread {

    private DatagramSocket socket;
    private boolean fin = false;
    private InetAddress ipServer;
    private int puertoServer;

    private final List<String> bandeja = new ArrayList<>();

    public HiloCliente(String ip, int puerto) {
        try {
            socket = new DatagramSocket();
            ipServer = InetAddress.getByName(ip);
            puertoServer = puerto;
        } catch (SocketException e) {
            e.printStackTrace();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    @Override
    public void run() {
        while (!fin) {
            byte[] datos = new byte[1024];
            DatagramPacket dp = new DatagramPacket(datos, datos.length);
            try {
                socket.receive(dp);
                String msg = new String(dp.getData(), 0, dp.getLength());
                synchronized (bandeja) {
                    bandeja.add(msg);
                }
            } catch (IOException e) {
                if (!fin) {
                    e.printStackTrace();
                }
            }
        }
    }

    public void enviarMensaje(String msg) {
        byte[] data = msg.getBytes();
        DatagramPacket dp = new DatagramPacket(data, data.length, ipServer, puertoServer);
        try {
            socket.send(dp);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    public List<String> drenar() {
        synchronized (bandeja) {
            List<String> copia = new ArrayList<>(bandeja);
            bandeja.clear();
            return copia;
        }
    }

    public void detener() {
        fin = true;
        if (socket != null && !socket.isClosed()) {
            socket.close();
        }
    }

package com.capuccinolovesballerina.game.red.hilos;

import java.io.IOException;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetSocketAddress;
import java.net.SocketException;
import java.util.HashSet;
import java.util.Set;

public class HiloServidor extends Thread {

    private static final int PUERTO = 9992;
    private static final int MAX_JUGADORES = 2;

    private DatagramSocket socket;
    private boolean fin = false;

    private final InetSocketAddress[] destino = new InetSocketAddress[MAX_JUGADORES];
    private final boolean[] conectado = new boolean[MAX_JUGADORES];
    private final float[][] pos = new float[MAX_JUGADORES][4];
    private final boolean[] enSalida = new boolean[MAX_JUGADORES];

    private int siguienteRol = 0;

    private final Set<Integer> cortados = new HashSet<>();
    private boolean palanca = false;
    private boolean puerta = false;
    private float espumaX = 0f;
    private float espumaY = 0f;
    private boolean hayEspuma = false;
    private int muerto = -1;
    private boolean victoria = false;

    public HiloServidor() {
        try {
            socket = new DatagramSocket(PUERTO);
        } catch (SocketException e) {
            e.printStackTrace();
        }
    }

    @Override
    public void run() {
        while (!fin) {
            byte[] datos = new byte[1024];
            DatagramPacket dp = new DatagramPacket(datos, datos.length);
            try {
                socket.receive(dp);
                procesarMensaje(dp);
            } catch (IOException e) {
                if (!fin) {
                    e.printStackTrace();
                }
            }
        }
    }

    private void procesarMensaje(DatagramPacket dp) {
        String msg = new String(dp.getData(), 0, dp.getLength());
        String[] p = msg.split("\\|", -1);
        if (p.length == 0) {
            return;
        }
        InetSocketAddress origen = (InetSocketAddress) dp.getSocketAddress();
        String tipo = p[0];

        try {
            switch (tipo) {
                case "JOIN":
                    manejarJoin(origen);
                    break;
                case "POS":
                    manejarPos(origen, p);
                    break;
                case "CORTE":
                    manejarCorte(origen, p);
                    break;
                case "ESPUMA":
                    manejarEspuma(origen, p);
                    break;
                case "PALANCA":
                    manejarPalanca(origen);
                    break;
                case "SALIDA":
                    manejarSalida(origen, p);
                    break;
                case "MUERTE":
                    manejarMuerte(origen);
                    break;
                case "REINICIO":
                    manejarReinicio(origen);
                    break;
                default:
                    break;
            }
        } catch (Exception e) {
            System.out.println("mensaje invalido: " + msg);
        }
    }

    private int buscarRol(InetSocketAddress origen) {
        for (int i = 0; i < MAX_JUGADORES; i++) {
            if (conectado[i] && origen.equals(destino[i])) {
                return i;
            }
        }
        return -1;
    }

    private void mandar(InetSocketAddress a, String msg) {
        byte[] data = msg.getBytes();
        DatagramPacket dp = new DatagramPacket(data, data.length, a);
        try {
            socket.send(dp);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private void broadcast(String msg) {
        for (int i = 0; i < MAX_JUGADORES; i++) {
            if (conectado[i]) {
                mandar(destino[i], msg);
            }
        }
    }

    private void manejarJoin(InetSocketAddress origen) {
        int ya = buscarRol(origen);
        if (ya >= 0) {
            mandar(origen, "ROL|" + ya);
            mandarSnapshot(origen, ya);
            return;
        }
        if (siguienteRol >= MAX_JUGADORES) {
            mandar(origen, "LLENO");
            return;
        }
        int rol = siguienteRol++;
        destino[rol] = origen;
        conectado[rol] = true;
        mandar(origen, "ROL|" + rol);
        mandarSnapshot(origen, rol);
    }

    private void mandarSnapshot(InetSocketAddress a, int rol) {
        for (Integer i : cortados) {
            mandar(a, "CORTE|" + i);
        }
        mandar(a, "PALANCA|" + palanca + "|" + puerta);
        if (hayEspuma) {
            mandar(a, "ESPUMA|" + espumaX + "|" + espumaY);
        }
        if (muerto >= 0) {
            mandar(a, "MUERTE|" + muerto);
        }
        if (victoria) {
            mandar(a, "VICTORIA");
        }
        int otro = 1 - rol;
        if (otro >= 0 && otro < MAX_JUGADORES && conectado[otro]) {
            boolean mirandoOtro = pos[otro][2] > 0.5f;
            boolean sueloOtro = pos[otro][3] > 0.5f;
            mandar(a, "POS|" + otro + "|" + pos[otro][0] + "|" + pos[otro][1] + "|" + mirandoOtro + "|" + sueloOtro);
        }
    }

    private void manejarPos(InetSocketAddress origen, String[] p) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        pos[rol][0] = Float.parseFloat(p[1]);
        pos[rol][1] = Float.parseFloat(p[2]);
        pos[rol][2] = Boolean.parseBoolean(p[3]) ? 1f : 0f;
        pos[rol][3] = Boolean.parseBoolean(p[4]) ? 1f : 0f;
        broadcast("POS|" + rol + "|" + p[1] + "|" + p[2] + "|" + p[3] + "|" + p[4]);
    }

    private void manejarCorte(InetSocketAddress origen, String[] p) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        int indice = Integer.parseInt(p[1]);
        cortados.add(indice);
        broadcast("CORTE|" + indice);
    }

    private void manejarEspuma(InetSocketAddress origen, String[] p) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        espumaX = Float.parseFloat(p[1]);
        espumaY = Float.parseFloat(p[2]);
        hayEspuma = true;
        broadcast("ESPUMA|" + espumaX + "|" + espumaY);
    }

    private void manejarPalanca(InetSocketAddress origen) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        palanca = !palanca;
        puerta = palanca;
        broadcast("PALANCA|" + palanca + "|" + puerta);
    }

    private void manejarSalida(InetSocketAddress origen, String[] p) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        enSalida[rol] = Boolean.parseBoolean(p[1]);
        if (!victoria && enSalida[0] && enSalida[1]) {
            victoria = true;
            broadcast("VICTORIA");
        }
    }

    private void manejarMuerte(InetSocketAddress origen) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        muerto = rol;
        broadcast("MUERTE|" + rol);
    }

    private void manejarReinicio(InetSocketAddress origen) {
        int rol = buscarRol(origen);
        if (rol < 0) {
            return;
        }
        cortados.clear();
        palanca = false;
        puerta = false;
        hayEspuma = false;
        muerto = -1;
        victoria = false;
        enSalida[0] = false;
        enSalida[1] = false;
        broadcast("REINICIO");
    }

    public void detener() {
        fin = true;
        if (socket != null && !socket.isClosed()) {
            socket.close();
        }
    }
}

package com.capuccinolovesballerina.game.Utilidades;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;

public class ControladorPartida {

    private final Nivel nivel;
    private final ElementosNivel elementos;

    private final MenuPausa menuPausa;
    private final MenuGameOver menuGameOver;
    private final MenuVictoria menuVictoria;
    private final HUD hud;

    private final ControladorEntradaJugador entrada;
    private final CapuccinoAssassino cappuccino;
    private final BallerinaCapuccina ballerina;
    private final AnimacionesCappuccino animaciones;
    private static final float ALCANCE_PALANCA = 12f;

    public ControladorPartida(
        Nivel nivel,
        ElementosNivel elementos,
        MenuPausa menuPausa,
        MenuGameOver menuGameOver,
        MenuVictoria menuVictoria,
        HUD hud,
        ControladorEntradaJugador entrada,
        CapuccinoAssassino cappuccino,
        BallerinaCapuccina ballerina,
        AnimacionesCappuccino animaciones
    ) {
        this.nivel = nivel;
        this.elementos = elementos;
        this.menuPausa = menuPausa;
        this.menuGameOver = menuGameOver;
        this.menuVictoria = menuVictoria;
        this.hud = hud;
        this.entrada = entrada;
        this.cappuccino = cappuccino;
        this.ballerina = ballerina;
        this.animaciones = animaciones;
    }

    public void actualizar(float delta) {

        if (entrada.isEscapePresionado()) {
            if (menuPausa.estaVisible()) {
                menuPausa.ocultar();
            } else if (!menuGameOver.estaVisible() && !menuVictoria.estaVisible()) {
                menuPausa.mostrar();
                entrada.soltarTodo();
            }
        }

        if (menuPausa.estaVisible()) {
            menuPausa.actualizar(delta);
            return;
        }

        if (menuGameOver.estaVisible()) {
            menuGameOver.actualizar(delta);
            return;
        }

        if (menuVictoria.estaVisible()) {
            menuVictoria.actualizar(delta);
            return;
        }

        if (Gdx.input.getInputProcessor() != entrada) {
            Gdx.input.setInputProcessor(entrada);
        }

        hud.actualizar(delta);

        ballerina.actualizarHabilidad(delta);

        if (entrada.isHabilidadPresionadaBallerina()) {
            ballerina.crearEspuma();
        }


        Array<Rectangle> solidos = new Array<>(nivel.getColisiones().getSolidos());

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (cortable.bloquea()) {
                solidos.add(cortable.getZona());
            }
        }

        BloqueEspuma espuma = ballerina.getEspuma();
        if (espuma != null) {
            solidos.add(espuma.getZona());
        }

        cappuccino.actualizar(
            delta,
            entrada.isIzquierda(),           // A
            entrada.isDerecha(),             // D
            entrada.isSaltoPresionado(),     // W / Espacio
            solidos
        );

        ballerina.actualizar(
            delta,
            entrada.isIzquierdaBallerina(),      // ←
            entrada.isDerechaBallerina(),        // →
            entrada.isSaltoPresionadoBallerina(),// ↑
            solidos
        );


        if (entrada.isHabilidadPresionada()) {
            animaciones.setEstado(Estado.ATTACK);
            cortar();
        } else if (!cappuccino.isEnSuelo()) {
            animaciones.setEstado(Estado.JUMP);
        } else if (entrada.isIzquierda() || entrada.isDerecha()) {
            animaciones.setEstado(Estado.WALK);
        } else {
            animaciones.setEstado(Estado.IDLE);
        }

        animaciones.actualizar(delta);


        if (estaMuerto(cappuccino) || estaMuerto(ballerina)) {
            mostrarGameOver();
            return;
        }


        comprobarVictoria();

        if (menuVictoria.estaVisible()) {
            return;
        }

        entrada.limpiarEventos();
    }

    private void actualizarPalancas() {
        Palanca palCap = elementos.getPalancaCappuccino();
        Palanca palBal = elementos.getPalancaBallerina();


        if (entrada.isInteractuarPresionado()) {
            Gdx.app.log("PALANCA", "E: objeto=" + (palCap != null)
                + " cerca=" + estaCerca(cappuccino, palCap));
        }
        if (entrada.isInteractuarPresionadoBallerina()) {
            Gdx.app.log("PALANCA", "ENTER: objeto=" + (palBal != null)
                + " cerca=" + estaCerca(ballerina, palBal));
        }
    }
    private boolean estaCerca(Player personaje, Palanca palanca) {
        if (palanca == null) return false;
        Rectangle c = personaje.getCaja();
        Rectangle alcance = new Rectangle(
            c.x - ALCANCE_PALANCA, c.y - ALCANCE_PALANCA,
            c.width + 2 * ALCANCE_PALANCA, c.height + 2 * ALCANCE_PALANCA);
        return alcance.overlaps(palanca.getZona());
    }
    private void intentarAlternar(Palanca palanca) {
        Puerta puerta = elementos.getPuerta();

        // No dejar cerrar la puerta con alguien parado adentro
        boolean seCerraria = palanca.estaActivada() && puerta != null && puerta.estaAbierta();
        if (seCerraria && alguienEnLaPuerta()) return;

        palanca.alternar();
        actualizarPuerta();
    }


    private void actualizarPuerta() {
        Puerta puerta = elementos.getPuerta();
        Palanca palCap = elementos.getPalancaCappuccino();
        Palanca palBal = elementos.getPalancaBallerina();
        if (puerta == null || (palCap == null && palBal == null)) return;

        // si un nivel solo tiene una palanca, la que falta no bloquea
        boolean capOk = palCap == null || palCap.estaActivada();
        boolean balOk = palBal == null || palBal.estaActivada();
        puerta.setAbierta(capOk && balOk);
    }
    private boolean alguienEnLaPuerta() {
        Puerta puerta = elementos.getPuerta();
        return puerta != null
            && (cappuccino.getCaja().overlaps(puerta.getZona())
            || ballerina.getCaja().overlaps(puerta.getZona()));
    }


    private void comprobarVictoria() {
        Puerta puerta = elementos.getPuerta();

        if (puerta == null || !puerta.estaAbierta()) {
            return;
        }

        Rectangle zona = puerta.getZona();

        if (cappuccino.getCaja().overlaps(zona) && ballerina.getCaja().overlaps(zona)) {
            entrada.soltarTodo();
            menuVictoria.mostrar();
        }
    }

    private boolean estaMuerto(Player personaje) {
        if (personaje.getY() < -personaje.getAlto()) {
            return true;
        }

        for (Rectangle peligro : elementos.getHitboxPeligros()) {
            if (personaje.getCaja().overlaps(peligro)) {
                return true;
            }
        }

        return false;
    }

    private void mostrarGameOver() {
        entrada.soltarTodo();
        menuGameOver.mostrar();
    }

    private void cortar() {
        Rectangle caja = cappuccino.getCaja();
        float alcance = 40f;
        Rectangle zonaCorte;

        if (cappuccino.isMirandoDerecha()) {
            zonaCorte = new Rectangle(caja.x + caja.width, caja.y, alcance, caja.height);
        } else {
            zonaCorte = new Rectangle(caja.x - alcance, caja.y, alcance, caja.height);
        }

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (!cortable.estaCortado() && zonaCorte.overlaps(cortable.getZona())) {
                cortable.setCortado(true);
            }
        }
    }

    public void reiniciar() {
        cappuccino.reaparecer(elementos.getSpawnX(), elementos.getSpawnY());
        ballerina.reaparecer(elementos.getSpawnBallerinaX(), elementos.getSpawnBallerinaY());

        ballerina.reiniciarHabilidad();

        Puerta puerta = elementos.getPuerta();

        if (puerta != null) {
            puerta.setAbierta(false);
        }

        Palanca palCap = elementos.getPalancaCappuccino();
        Palanca palBal = elementos.getPalancaBallerina();
        if (palCap != null && palCap.estaActivada()) palCap.alternar();
        if (palBal != null && palBal.estaActivada()) palBal.alternar();

        for (ObjetoCortable cortable : elementos.getCortables()) {
            cortable.setCortado(false);
        }

        entrada.soltarTodo();
        animaciones.reiniciar();
        hud.reiniciar();
    }
}

package com.capuccinolovesballerina.game.Utilidades;

import java.util.InputMismatchException;
import java.util.Scanner;

public class Entrada {

    private Scanner s = new Scanner(System.in);

    public int ingresarEntero(int min, int max) {
        boolean error;
        int numero = 0;

        do {
            error = false;

            try {
                numero = s.nextInt();

                if (numero > max || numero < min) {
                    System.out.println(
                        "Ingrese un número entre " + min + " y " + max
                    );
                    error = true;
                }

            } catch (InputMismatchException e) {
                System.out.println("Ingrese un valor numérico");
                error = true;

            } finally {
                s.nextLine();
            }

        } while (error);

        return numero;
    }

    public String ingresarTexto() {
        return s.nextLine();
    }
}

    private Texture texturaPalancaOff;
    private Texture texturaPalancaOn;

    private Texture texturaLiana;

    public RenderizadorJuego(
        OrthographicCamera camera,
        Nivel nivel,
        CapuccinoAssassino cappuccino,
        BallerinaCapuccina ballerina,
        AnimacionesCappuccino animaciones,
        ElementosNivel elementos,

        this.elementos = elementos;
        this.hud = hud;
        this.menuPausa = menuPausa;

        this.menuVictoria = menuVictoria;

        shapeRenderer = new ShapeRenderer();
        batch = new SpriteBatch();

    private void cargarTexturas() {
        texturaCerrada = new Texture("objetos/puerta_cerrada.png");
        texturaAbierta = new Texture("objetos/puerta_abierta.png");
        texturaPalancaOff = new Texture("objetos/palanca_apagada.png");
        texturaPalancaOn = new Texture("objetos/palanca_encendida.png");
        texturaPinchos = new Texture("objetos/pinchos.png");
        texturaLiana = new Texture("objetos/liana.png");

        texturaCerrada.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
        texturaAbierta.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
        texturaPalancaOff.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
        texturaPalancaOn.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);

        texturaLiana.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);

    public void dibujar() {

        ScreenUtils.clear(0.1f, 0.1f, 0.15f, 1f);

        camera.update();
        renderizador.setView(camera);
        renderizador.render();

        dibujarCortables();
        dibujarEspuma();

        batch.setProjectionMatrix(camera.combined);
        batch.begin();

        dibujarPuerta();
        dibujarPalanca(elementos.getPalancaCappuccino(), new Color(0.6f, 0.8f, 1f, 1f));
        dibujarPalanca(elementos.getPalancaBallerina(), new Color(1f, 0.6f, 0.8f, 1f));
        dibujarPeligros();
        dibujarLianas();
        dibujarPersonaje();

        batch.end();

        dibujarBallerinaPlaceholder();

        hud.dibujar();
        menuPausa.dibujar();
        menuGameOver.dibujar();
        menuVictoria.dibujar();
    }

    private void dibujarCortables() {
        shapeRenderer.setProjectionMatrix(camera.combined);
        shapeRenderer.begin(ShapeRenderer.ShapeType.Filled);

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (cortable.estaCortado()) continue;
            if ("liana".equals(cortable.getTipo())) continue;

            if ("cuerda".equals(cortable.getTipo())) {
                shapeRenderer.setColor(0.8f, 0.7f, 0.4f, 1f);
            } else {
                shapeRenderer.setColor(0.55f, 0.35f, 0.15f, 1f);
            }

            Rectangle z = cortable.getZona();
            shapeRenderer.rect(z.x, z.y, z.width, z.height);
        }

        shapeRenderer.end();

        BloqueEspuma espuma = ballerina.getEspuma();
        if (espuma == null) return;

        shapeRenderer.setProjectionMatrix(camera.combined);
        shapeRenderer.begin(ShapeRenderer.ShapeType.Filled);
        shapeRenderer.setColor(1f, 1f, 1f, espuma.getAlpha());

        Rectangle z = espuma.getZona();
        shapeRenderer.rect(z.x, z.y, z.width, z.height);

        shapeRenderer.end();
    }

        if (puerta == null) return;

        Rectangle z = puerta.getZona();
        Texture textura = puerta.estaAbierta() ? texturaAbierta : texturaCerrada;

        float ancho = 96f;
        float alto = 128f;
        batch.draw(textura, z.x + (z.width - ancho) / 2f, z.y, ancho, alto);
    }

    private void dibujarPalanca(Palanca palanca, Color tinte) {
        if (palanca == null) return;

        Rectangle z = palanca.getZona();

        float ancho = Constantes.ANCHO_PALANCA;   // antes: 48f

        batch.setColor(tinte);
        batch.draw(textura, z.x + (z.width - ancho) / 2f, z.y, ancho, alto);
        batch.setColor(Color.WHITE);
    }

    private void dibujarPeligros() {

            batch.draw(texturaPinchos, peligro.x, peligro.y, peligro.width, peligro.width / 2f);

/**
 * A helper object for game startup, featuring three utilities related to LWJGL3 on various operating systems.
 * <p>
 * The utilities are as follows:
 * <ul>
 *  <li> Windows: Prevents a common crash related to LWJGL3's extraction of shared library files.</li>
 *  <li> macOS: Spawns a child JVM process with {@code -XstartOnFirstThread} in the JVM args (if it was not already).
 *  This is required for LWJGL3 to work on macOS.</li>
 *  <li> Linux (NVIDIA GPUs only): Spawns a child JVM process with the {@code __GL_THREADED_OPTIMIZATIONS}
 *  {@link System#getenv(String) Environment Variable} set to {@code 0} (if it was not already). This is required for
 *  LWJGL3 to work on Linux with NVIDIA GPUs.</li>
 * </ul>
 * <a href="https://jvm-gaming.org/t/starting-jvm-on-mac-with-xstartonfirstthread-programmatically/57547">Based on this java-gaming.org post by kappa</a>
 * @author damios
 */
public class StartupHelper {

	private StartupHelper() {}

	private static final String JVM_RESTARTED_ARG = "jvmIsRestarted";

	/**
	 * Must only be called on Linux. Check OS first (or use short-circuit evaluation)!
	 * @return whether NVIDIA drivers are present on Linux.
	 */
	public static boolean isLinuxNvidia() {
		String[] drivers = new File("/proc/driver").list(
			(dir, path) -> path.toUpperCase(Locale.ROOT).contains("NVIDIA")
		);
		if (drivers == null) return false;
		return drivers.length > 0;
	}

	/**
	 * Applies the utilities as described by {@link StartupHelper}'s Javadoc.
	 * <p>
	 * All {@link System#getenv() Environment Variables} are copied to the child JVM process (if it is spawned), as
	 * specified by {@link ProcessBuilder#environment()}; the same applies for
	 * {@link System#getProperties() System Properties}.
	 * <p>
	 * <b>Usage:</b>
	 * <pre><code>
	 * public static void main(String[] args) {
	 * 	 if (StartupHelper.startNewJvmIfRequired()) return;
	 * 	 // ... The rest of main() goes here, as normal.
	 * }
	 * </code></pre>
	 * @return whether a child JVM process was spawned or not.
	 */
	public static boolean startNewJvmIfRequired() {
		return startNewJvmIfRequired(true);
	}

	/**
	 * Applies the utilities as described by {@link StartupHelper}'s Javadoc.
	 * <p>
	 * All {@link System#getenv() Environment Variables} are copied to the child JVM process (if it is spawned), as
	 * specified by {@link ProcessBuilder#environment()}; the same applies for
	 * {@link System#getProperties() System Properties}.
	 * <p>
	 * <b>Usage:</b>
	 * <pre><code>
	 * public static void main(String[] args) {
	 *   // The parameter on the next line could instead be false if you don't want to inherit IO.
	 * 	 if (StartupHelper.startNewJvmIfRequired(true)) return;
	 * 	 // ... The rest of main() goes here, as normal.
	 * }
	 * </code></pre>
	 * @param inheritIO whether I/O should be inherited in the child JVM process. Please note that enabling this will
	 *                  block the thread until the child JVM process stops executing.
	 * @return whether a child JVM process was spawned or not.
	 */
	public static boolean startNewJvmIfRequired(boolean inheritIO) {
		String osName = System.getProperty("os.name").toLowerCase(Locale.ROOT);
		if (osName.contains("mac")) return startNewJvm0(/*isMac =*/ true, inheritIO);
		if (osName.contains("windows")) {
			// Here, we are trying to work around an issue with how LWJGL3 loads its extracted .dll files.
			// By default, LWJGL3 extracts to the directory specified by "java.io.tmpdir": usually, the user's home.
			// If the user's name has non-ASCII (or some non-alphanumeric) characters in it, that would fail.
			// By extracting to the relevant "ProgramData" folder, which is usually "C:\ProgramData", we avoid this.
			// We also temporarily change the "user.name" property to one without any chars that would be invalid.
			// We revert our changes immediately after loading LWJGL3 natives.
			String programData = System.getenv("ProgramData");
			if (programData == null) programData = "C:\\Temp"; // if ProgramData isn't set, try some fallback.
			String prevTmpDir = System.getProperty("java.io.tmpdir", programData);
			String prevUser = System.getProperty("user.name", "libGDX_User");
			System.setProperty("java.io.tmpdir", programData + "\\libGDX-temp");
			System.setProperty(
				"user.name",
				("User_" + prevUser.hashCode() + "_GDX" + Version.VERSION).replace('.', '_')
			);
			Lwjgl3NativesLoader.load();
			System.setProperty("java.io.tmpdir", prevTmpDir);
			System.setProperty("user.name", prevUser);
			return false;
		}
		return startNewJvm0(/*isMac =*/ false, inheritIO);
	}

	private static final String MAC_JRE_ERR_MSG = "A Java installation could not be found. If you are distributing this app with a bundled JRE, be sure to set the '-XstartOnFirstThread' argument manually!";
	private static final String LINUX_JRE_ERR_MSG = "A Java installation could not be found. If you are distributing this app with a bundled JRE, be sure to set the environment variable '__GL_THREADED_OPTIMIZATIONS' to '0'!";
	private static final String CHILD_LOOP_ERR_MSG = "The current JVM process is a spawned child JVM process, but StartupHelper has attempted to spawn another child JVM process! This is a broken state, and should not normally happen! Your game may crash or not function properly!";

	/**
	 * Spawns a child JVM process if on macOS, or on Linux with NVIDIA drivers.
	 * <p>
	 * All {@link System#getenv() Environment Variables} are copied to the child JVM process (if it is spawned), as
	 * specified by {@link ProcessBuilder#environment()}; the same applies for
	 * {@link System#getProperties() System Properties}.
	 * @param isMac whether the current OS is macOS. If this is `false` then the current OS is assumed to be Linux (and
	 *             an immediate check for NVIDIA drivers is performed).
	 * @param inheritIO whether I/O should be inherited in the child JVM process. Please note that enabling this will
	 *                 block the thread until the child JVM process stops executing.
	 * @return whether a child JVM process was spawned or not.
	 */
	public static boolean startNewJvm0(boolean isMac, boolean inheritIO) {
		long processID = getProcessID(isMac);
		if (!isMac) {
			// No need to restart non-NVIDIA Linux
			if (!isLinuxNvidia()) return false;
			// check whether __GL_THREADED_OPTIMIZATIONS is already disabled
			if ("0".equals(System.getenv("__GL_THREADED_OPTIMIZATIONS"))) return false;
		} else {
			// There is no need for -XstartOnFirstThread on Graal native image
			if (!System.getProperty("org.graalvm.nativeimage.imagecode", "").isEmpty()) return false;

			// Checks if we are already on the main thread, such as from running via Construo.
			long objcMsgSend = ObjCRuntime.getLibrary().getFunctionAddress("objc_msgSend");
			long nsThread = ObjCRuntime.objc_getClass("NSThread");
			long currentThread = JNI.invokePPP(nsThread, ObjCRuntime.sel_getUid("currentThread"), objcMsgSend);
			boolean isMainThread = JNI.invokePPZ(currentThread, ObjCRuntime.sel_getUid("isMainThread"), objcMsgSend);
			if (isMainThread) return false;

			if ("1".equals(System.getenv("JAVA_STARTED_ON_FIRST_THREAD_" + processID))) return false;
		}

		// Check whether this JVM process is a child JVM process already.
		// This state shouldn't usually be reachable, but this stops us from endlessly spawning new child JVM processes.
		if ("true".equals(System.getProperty(JVM_RESTARTED_ARG))) {
			System.err.println(CHILD_LOOP_ERR_MSG);
			return false;
		}

		// Spawn the child JVM process with updated environment variables or JVM args
		List<String> jvmArgs = new ArrayList<>();
		// The following line is used assuming you target Java 8, the minimum for LWJGL3.
		String javaExecPath = System.getProperty("java.home") + "/bin/java";
		// If targeting Java 9 or higher, you could use the following instead of the above line:
		//String javaExecPath = ProcessHandle.current().info().command().orElseThrow()
		if (!(new File(javaExecPath).exists())) {
			System.err.println(getJreErrMsg(isMac));
			return false;
		}

		jvmArgs.add(javaExecPath);
		if (isMac) jvmArgs.add("-XstartOnFirstThread");
		jvmArgs.add("-D" + JVM_RESTARTED_ARG + "=true");
		jvmArgs.addAll(ManagementFactory.getRuntimeMXBean().getInputArguments());
		jvmArgs.add("-cp");
		jvmArgs.add(System.getProperty("java.class.path"));
		String mainClass = System.getenv("JAVA_MAIN_CLASS_" + processID);
		if (mainClass == null) {
			StackTraceElement[] trace = Thread.currentThread().getStackTrace();
			if (trace.length > 0) mainClass = trace[trace.length - 1].getClassName();
			else {
				System.err.println("The main class could not be determined.");
				return false;
			}
		}
		jvmArgs.add(mainClass);

		try {
			ProcessBuilder processBuilder = new ProcessBuilder(jvmArgs);
			if (!isMac) processBuilder.environment().put("__GL_THREADED_OPTIMIZATIONS", "0");

			if (!inheritIO) processBuilder.start();
			else processBuilder.inheritIO().start().waitFor();
		} catch (Exception e) {
			System.err.println("There was a problem restarting the JVM.");
			// noinspection CallToPrintStackTrace
			e.printStackTrace();
		}

		return true;
	}

	private static String getJreErrMsg(boolean isMac) {
		if (isMac) return MAC_JRE_ERR_MSG;
		else return LINUX_JRE_ERR_MSG;
	}

	private static long getProcessID(boolean isMac) {
		if (isMac) return LibC.getpid();
		else return UNISTD.getpid();
	}
}

# Capuccino Loves Ballerina

## Integrantes del grupo

- Guillermo Bordoli
- Agustín Bruno
- Santino Fierro

## Descripción del videojuego

Capuccino Loves Ballerina es un videojuego cooperativo de plataformas y puzles en 2D para PC, inspirado en las mecánicas principales de Fireboy and Watergirl.

Cada jugador controlará desde una computadora diferente a Cappuccino Assassino o Ballerina Capuccina. Ambos personajes deberán colaborar para superar obstáculos, activar mecanismos, resolver puzles y escapar de una estructura misteriosa mediante el uso coordinado de sus habilidades especiales.

## Tecnologías principales

- Java JDK 17
- LibGDX 1.14.1
- Gradle
- IntelliJ IDEA
- Git y GitHub
- Plataforma objetivo: escritorio mediante LWJGL3

## Wiki del proyecto

La propuesta formal y detallada del videojuego se encuentra en la Wiki:

[Ver la propuesta del proyecto](https://github.com/agustinbruno07/capuccino-loves-ballerina/wiki)

## Compilación y ejecución

### Requisitos

Para compilar y ejecutar el proyecto es necesario contar con:

- Java JDK 17.
- Git.
- IntelliJ IDEA o cualquier entorno compatible con proyectos Gradle.

### Clonar el repositorio

Abrir una terminal y ejecutar:

```bash
git clone https://github.com/agustinbruno07/capuccino-loves-ballerina.git
```

### Ingresar a la carpeta del proyecto

Una vez clonado el repositorio, ejecutar:

```bash
cd capuccino-loves-ballerina
```

### Ejecutar en Windows

Desde la carpeta raíz del proyecto, ejecutar:

```bash
gradlew.bat lwjgl3:run
```

Si se utiliza Git Bash, también se puede ejecutar:

```bash
./gradlew.bat lwjgl3:run
```

### Ejecutar en Linux o macOS

Desde la carpeta raíz del proyecto, ejecutar:

```bash
./gradlew lwjgl3:run
```

### Ejecutar desde IntelliJ IDEA

1. Abrir IntelliJ IDEA.
2. Seleccionar `Open`.
3. Buscar y abrir la carpeta `capuccino-loves-ballerina`.
4. Esperar a que Gradle termine de importar el proyecto y descargar las dependencias.
5. Abrir el módulo `lwjgl3`.
6. Buscar la clase `Lwjgl3Launcher.java`.
7. Ejecutar el método `main` de `Lwjgl3Launcher`.

## [1.6.0] - 2026-11-07

### Agregado
- **Infraestructura de Red UDP:**
    - Clase `Servidor` y `HiloServidor`: Escucha puerto fijo, gestiona máx. 2 jugadores, asigna roles automáticos, mantiene estado autoritativo (palanca/puerta/cortes/muertes) y envía snapshots de bienvenida.
    - Clase `Cliente` y `HiloCliente`: Conexión asíncrona, cola de recepción segura (`synchronized`), métodos helpers para enviar inputs específicos (`enviarPosicion`, `enviarCorte`, etc.).

### Corregido
- Bug crítico en `HiloServidor.mandarSnapshot`: Los flags booleanos (`mirando`, `enSuelo`) ahora se serializan como `"true"/"false"` consistentes con el broadcast en vivo, previniendo errores de parsing (`Boolean.parseBoolean("1.0") == false`) en clientes reconectados.

### Notas Técnicas
- Validación realizada mediante smoke test de 3 instancias locales simulando conexión secuencial y saturación de cupos.
- La integración final con `ControladorPartida` y `PantallaJuego` queda pendiente para la siguiente iteración.

- `reintentar()` ahora también devuelve la puerta y la palanca a su estado inicial, además de reposicionar al personaje.
